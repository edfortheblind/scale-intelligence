How to: Create a Detail Pane Stored Procedure
View with Navigation Tools
User Interface Extensibility > Metadata Web Pages > Insight Pages > Detail Pane > How to: Create a Detail Pane Stored Procedure

The GenericDataBind web service can be used to retrieve data from a stored procedure call and return it. Also see the How to: Bind Detail Pane Controls topic for more information on getting the GenericDataBind web service to call the proper stored procedure and binding the data to the controls. The stored procedure can return several different types of data as noted in the table below.

Stored Procedure Section Description
SCALAR Tables returned from the stored procedure that are single row data which is intended to bind with labels or non-array structures should have the first column named "SCALAR". The other column names in this type of table will be used to bind with controls in the detail pane.
TABLE Tables returned from the stored procedure that will bind to an array or table structure should have the first column named "Table_<ControlName>". The other column names in this type of table serve as resource keys that translate to the specified culture and are shown as grid header text.
TEMPLATE Tables returned from the stored procedure that will bind to a templated control should have the first column named "Template_<ControlName>".
HIERARCHICAL Tables returned from the stored procedure that will bind to a hierarchical grid should have the first column named "Hierarchical_<ControlName>". The other column names in this type of table serve as resource keys that translate to the specified culture and are shown as grid header text.

Example GenericDataBind Stored Procedure
Copy Code 
CREATE PROCEDURE SHP_InsightDetailPaneData(@internalShipmentNum int , @culture nvarchar(10)) 
AS
BEGIN
declare @dockDoorInternalNum numeric(9,0);
select * INTO #tempShipment
FROM SHIPMENT_HEADER_VIEW
WHERE INTERNAL_SHIPMENT_NUM = @internalShipmentNum;
SELECT top 1 'SCALAR' AS SCALAR,
    sh.SHIPMENT_ID AS ShipmentID,
    sh.INTERNAL_SHIPMENT_NUM AS InternalShipmentNum,
    sh.ORDER_TYPE AS OrderType ,
    sh.CUSTOMER_name AS Customer,
    sh.ship_to_name AS ShipTo,
    sh.CARRIER AS Carrier ,
    sh.CARRIER_SERVICE AS CarrierService,
    dbo.STSfn_RtrvStsName('outbound' ,sh.TRAILING_STS) AS TrailingSts ,
    dbo.STSfn_RtrvStsName ('outbound' ,sh.LEADING_STS) AS LeadingSts ,
    TOTAL_LINES as SummaryDetails,
    TOTAL_QTY as SummaryUnits,
    TOTAL_VALUE as SummaryValue,
    TOTAL_CONTAINERS as SummaryContainers
FROM #tempShipment sh;
SELECT @dockDoorInternalNum=sl.DOCK_DOOR
FROM #tempShipment sh
LEFT OUTER JOIN Shipping_load sl ON sh.SHIPPING_LOAD_NUM = sl.INTERNAL_LOAD_NUM;
SELECT top 1 'SCALAR' AS SCALAR,
    LOCATION AS DockDoor,
    LOCATION AS LoadDockDoor
FROM LOCATION
WHERE OBJECT_ID=@dockDoorInternalNum;
SELECT 'Template_DetailPaneCustomerAddress' AS Template_DetailPaneCustomerAddress ,
    sh.CUSTOMER_ATTENTION_TO AS AttentionTo,
    sh.customer AS BeforeName ,
    sh.CUSTOMER_NAME AS Name,
    sh.CUSTOMER_ADDRESS1 AS Address1,
    sh.CUSTOMER_ADDRESS2 AS Address2,
    sh.CUSTOMER_ADDRESS3 AS Address3,
    sh.CUSTOMER_CITY AS City,
    sh.customer_state AS [State],
    gcd.[DESCRIPTION] AS Country,
    sh.CUSTOMER_POSTAL_CODE AS PostalCode,
    sh.CUSTOMER_PHONE_NUM AS Phone ,
    sh.CUSTOMER_FAX_NUM AS Fax,
    sh.CUSTOMER_EMAIL_ADDRESS AS Email
FROM #tempShipment sh LEFT OUTER JOIN
GENERIC_CONFIG_DETAIL gcd on sh.CUSTOMER_COUNTRY = gcd.IDENTIFIER AND gcd.RECORD_TYPE='Country';
SELECT 'Template_DetailPaneShipToAddress' AS Template_DetailPaneShipToAddress ,
    sh.Ship_To_ATTENTION_TO AS AttentionTo,
    sh.Ship_To AS BeforeName ,
    sh.Ship_To_NAME AS Name,
    sh.Ship_To_ADDRESS1 AS Address1,
    sh.Ship_To_ADDRESS2 AS Address2,
    sh.Ship_To_ADDRESS3    AS Address3,
    sh.Ship_To_CITY AS City,
    sh.Ship_To_state AS [State],
    gcd.[DESCRIPTION] AS Country,
    sh.Ship_To_POSTAL_CODE AS PostalCode,
    sh.Ship_To_PHONE_NUM AS Phone ,
    sh.Ship_To_FAX_NUM AS Fax,
    sh.Ship_To_EMAIL_ADDRESS AS Email
FROM #tempShipment sh LEFT OUTER JOIN
GENERIC_CONFIG_DETAIL gcd on sh.SHIP_TO_COUNTRY = gcd.IDENTIFIER AND gcd.RECORD_TYPE='Country';
select 'Table_DetailPaneDetails' AS Table_DetailPaneDetails,
    ITEM as ITEM,
    [ITEM_DESC] as [DESCRIPTION],
    TOTAL_QTY as TOTALQTY#30,
    QUANTITY_UM as UM
FROM SHIPMENT_DETAIL sd
WHERE sd.INTERNAL_SHIPMENT_NUM = @internalShipmentNum;
SELECT 'Table_DetailPaneDetailsUserDef' AS Table_DetailPaneDetailsUserDef,
    USER_DEF1 as UDSHH1,
    USER_DEF2 as UDSHH2,
    USER_DEF3 as UDSHH3,
    USER_DEF4 AS UDSHH4,
    USER_DEF5 AS UDSHH5,
    USER_DEF6 AS UDSHH6,
    USER_DEF7 AS UDSHH7#40,
    USER_DEF8 AS UDSHH8#40
FROM #tempShipment;
select top 1 'Table_DetailPaneDetailsStatusInfo' AS Table_DetailPaneDetailsStatusInfo,
    dbo.STSfn_RtrvStsName ('outbound',LEADING_STS) as LEADINGSTS,
    dbo.STSfn_RtrvStsName ('outbound',TRAILING_STS) as TRAILINGSTS,
    PRIORITY as [PRIORITY],
    REJECTION_NOTE as REJECTIONNOTE,
    CASE WHEN IMMEDIATE_NEEDS_NOTE is NOT NULL THEN dbo.RSCMfn_RtrvResource(IMMEDIATE_NEEDS_NOTE,'msg',@culture)
    ELSE IMMEDIATE_NEEDS_NOTE END AS IMMEDIATENEEDSNOTE
FROM #tempShipment;
select top 1 'Table_DetailPaneDetailsCarrierInfo' AS Table_DetailPaneDetailsCarrierInfo,
    CARRIER AS CARRIER,
    CARRIER_SERVICE AS CARRIERSERVICE,
    CARRIER_GROUP AS CARRIERGROUP,
    TOTAL_FREIGHT_CHARGE AS TOTALFREIGHT#40,
    FREIGHT_TERMS AS FREIGHTTERMS,
    BOL_NUM_ALPHA AS BOLNUMBER,
    PRO_NUM_ALPHA AS PRONUMBER,
    ROUTE AS [ROUTE]
FROM #tempShipment;
select top 1 'Table_DetailPaneDetailsDates' AS Table_DetailPaneDetailsDates,
    PLANNED_SHIP_DATE AS PLANNEDSHIPDATE ,
    SCHEDULED_SHIP_DATE AS SCHEDULEDSHIPDATE,
    ACTUAL_SHIP_DATE_TIME AS ACTUALSHIPDATE,
    CASE WHEN REQUESTED_DELIVERY_TYPE is NOT NULL THEN dbo.RSCMfn_RtrvResource('DELIVER'+REQUESTED_DELIVERY_TYPE,'text',@culture)
    ELSE REQUESTED_DELIVERY_TYPE END AS REQUESTEDDELIVERYTYPE ,
    PLANNED_DELIVERY_DATE_TIME AS PLANNEDDELIVERYDATE,
    ACTUAL_DELIVERY_DATE_TIME AS ACTUALDELIVERYDATETIME,
    REQUESTED_DELIVERY_DATE AS REQUESTEDDELIVERYDATE
FROM #tempShipment;
SELECT 'Hierarchical_DetailPaneDetailsContainers' AS Hierarchical_DetailPaneDetailsContainers,
    INTERNAL_CONTAINER_NUM AS META_ID,
    PARENT AS META_PARENTID ,
    CONTAINER_ID AS ID,
    CONTAINER_TYPE AS [TYPE],
    dbo.STSfn_RtrvStsName ('outbound',[STATUS]) AS [STATUS],
    dbo.SHIPCONTfn_RtrvCurrentLocation(INTERNAL_CONTAINER_NUM) AS LOCATION,
    INTERNAL_CONTAINER_NUM AS HIDDEN_INTERNALCONTAINERNUM
FROM SHIPPING_CONTAINER
WHERE INTERNAL_SHIPMENT_NUM= @INTERNALSHIPMENTNUM AND CONTAINER_ID IS NOT NULL;
select top 1 'Table_DetailPaneDetailsLoadInfo' AS Table_DetailPaneDetailsLoadInfo,
    sl.INTERNAL_LOAD_NUM as LOADNUMBER,
    sl.CARRIER as CARRIER,
    sl.SCHEDULED_SHIP_DATE AS SCHEDULEDSHIPDATE,
    dbo.STSfn_RtrvStsName ('outbound',sl.LEADING_STS) as LEADINGSTS,
    dbo.STSfn_RtrvStsName ('outbound',sl.TRAILING_STS) as TRAILINGSTS
FROM #tempShipment sh
LEFT OUTER JOIN Shipping_load sl ON sh.SHIPPING_LOAD_NUM = sl.INTERNAL_LOAD_NUM;
SELECT top 1 'Table_DetailPaneDetailsReferenceInfo' AS Table_DetailPaneDetailsReferenceInfo,
    ERP_ORDER                    AS ERPORDER,
    INTERNAL_SHIPMENT_NUM        AS INTERNALSHIPMENTNUM ,
    INTERNAL_ORDER_NUM            AS INTERNALORDERNUM,
    STORE_DISTRIBUTION            AS STOREDISTRIBUTION,
    ORDER_TYPE                    AS ORDERTYPE,
    PROCESS_TYPE                AS PROCESSTYPE,
    (CASE WHEN CONSOLIDATION_DOCK_LOC_AREA IS NOT NULL THEN (CONSOLIDATION_DOCK_LOC_AREA + N'-' + CONSOLIDATION_DOCK_LOC_POS) ELSE NULL END)    AS CONSOLIDATIONLOCATION,
    CONVERT(BIT, (CASE WHEN ISNULL(ALLOCATE_COMPLETE, N'N')= N'Y' THEN 1 ELSE 0 END))            AS ALLOCATECOMPLETE,
    CONVERT(BIT, (CASE WHEN ISNULL(CONSOLIDATION_ALLOWED, N'N')= N'Y' THEN 1 ELSE 0 END))        AS CONSOLIDATIONALLOWED,
    CONVERT(BIT, (CASE WHEN ISNULL(CONSOLIDATED, N'N')= N'Y' THEN 1 ELSE 0 END))                AS CONSOLIDATED,
    CONVERT(BIT, (CASE WHEN ISNULL(MANUALLY_ENTERED, N'N')= N'Y' THEN 1 ELSE 0 END))            AS MANUALLYENTERED
FROM #tempShipment;
END

These stored procedures require the "culture" parameter to properly translate resources.
