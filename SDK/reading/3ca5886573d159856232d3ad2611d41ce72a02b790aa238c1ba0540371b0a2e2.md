    <div id="i-body-content" class="i-body-content" data-source-node="n46">
        <p data-source-node="n47">Restrictions represent the list of fields on a screen that should always be restricted (or read-only). The system provides the exit point <em data-source-node="n48">Field Restrictions – Modify</em>, that should be used to override the list of base restrictions. The sql utility function <em data-source-node="n49">ModifyCommaSeparatedStringList</em> can be used to modify field restrictions in stored procedure exit point programs.</p>

<p data-source-node="n50"> </p>

<p data-source-node="n51">The following example represents a sample implementation of the <em data-source-node="n52">Field Restrictions – Modify</em> exit point program using a stored procedure. </p>

    <span style="COLOR: #008000" data-source-node="n67">@SESSIONVALUE</span> xml,
    <span style="COLOR: #008000" data-source-node="n68">@FORMID</span> numeric,
    <span style="COLOR: #008000" data-source-node="n69">@SCREENNAME</span> <span style="COLOR: #0000ff" data-source-node="n70">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n71">max</span>),

    <span style="COLOR: #008000" data-source-node="n73">@RESTRICTIONS</span> <span style="COLOR: #0000ff" data-source-node="n74">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n75">max</span>),
    <span style="COLOR: #008000" data-source-node="n76">@RETURNVALUE</span> <span style="COLOR: #0000ff" data-source-node="n77">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n78">max</span>) output
)
<span style="COLOR: #0000ff" data-source-node="n79">AS</span>

    <span style="COLOR: #0000ff" data-source-node="n82">DECLARE</span> <span style="COLOR: #008000" data-source-node="n83">@fieldsToAdd</span> <span style="COLOR: #0000ff" data-source-node="n84">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n85">max</span>), <span style="COLOR: #008000" data-source-node="n86">@fieldsToRemove</span> <span style="COLOR: #0000ff" data-source-node="n87">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n88">max</span>), <span style="COLOR: #008000" data-source-node="n89">@userName</span> <span style="COLOR: #0000ff" data-source-node="n90">nvarchar</span>(<span style="COLOR: #ff00ff" data-source-node="n91">max</span>);
    <span style="COLOR: #0000ff" data-source-node="n92">SET</span> <span style="COLOR: #008000" data-source-node="n93">@userName</span> <span style="COLOR: #808080" data-source-node="n94">=</span> <span style="COLOR: #ff0000" data-source-node="n95">''</span>;
    <span style="COLOR: #0000ff" data-source-node="n96">SET</span> <span style="COLOR: #008000" data-source-node="n97">@fieldsToAdd</span> <span style="COLOR: #808080" data-source-node="n98">=</span> <span style="COLOR: #ff0000" data-source-node="n99">''</span>;
    <span style="COLOR: #0000ff" data-source-node="n100">SET</span> <span style="COLOR: #008000" data-source-node="n101">@fieldsToRemove</span> <span style="COLOR: #808080" data-source-node="n102">=</span> <span style="COLOR: #ff0000" data-source-node="n103">''</span>;
    <span style="COLOR: #0000ff" data-source-node="n104">IF</span> (<span style="COLOR: #008000" data-source-node="n105">@FORMID</span> <span style="COLOR: #808080" data-source-node="n106">=</span> <span style="FONT-WEIGHT: bold;COLOR: #800000" data-source-node="n107">2759</span>) <span style="COLOR: #008080" data-source-node="n108">-- Work instruction
</span>    <span style="COLOR: #0000ff" data-source-node="n109">BEGIN</span>       
       
        <span style="COLOR: #008080" data-source-node="n110">-- If we want allow only a particular user to edit fields Description and User Defined 8
</span>        <span style="COLOR: #008080" data-source-node="n111">-- If user is not 'userName' then do not allow user to edit Description and User Defined 8 field
</span>        <span style="COLOR: #0000ff" data-source-node="n112">EXEC</span> dbo.GetXMLAttributeValueByAttributeName <span style="COLOR: #008000" data-source-node="n113">@SESSIONVALUE</span>,<span style="COLOR: #ff0000" data-source-node="n114">'Session/User'</span>,<span style="COLOR: #008000" data-source-node="n115">@userName</span> OUTPUT
        <span style="COLOR: #0000ff" data-source-node="n116">IF</span> (<span style="COLOR: #008000" data-source-node="n117">@userName</span> <span style="COLOR: #808080" data-source-node="n118">&lt;&gt;</span> <span style="COLOR: #ff0000" data-source-node="n119">'userName'</span>)
        <span style="COLOR: #0000ff" data-source-node="n120">BEGIN</span>
            <span style="COLOR: #0000ff" data-source-node="n121">SET</span> <span style="COLOR: #008000" data-source-node="n122">@fieldsToAdd</span> <span style="COLOR: #808080" data-source-node="n123">=</span> <span style="COLOR: #ff0000" data-source-node="n124">'ITEMDESCRIPTION,UDWORK_INSTRUCTION8'</span>
        <span style="COLOR: #0000ff" data-source-node="n125">END</span>

</span>        <span style="COLOR: #0000ff" data-source-node="n127">SELECT</span> <span style="COLOR: #008000" data-source-node="n128">@RETURNVALUE</span> <span style="COLOR: #808080" data-source-node="n129">=</span> dbo.ModifyCommaSeparatedStringList(<span style="COLOR: #008000" data-source-node="n130">@RESTRICTIONS</span>, <span style="COLOR: #008000" data-source-node="n131">@fieldsToAdd</span>, <span style="COLOR: #008000" data-source-node="n132">@fieldsToRemove</span>)
    <span style="COLOR: #0000ff" data-source-node="n133">END</span>
