Movement Class Analysis data mining functionality pulls data from the TRANSACTION_HISTORY and AR_TRANSACTION_HISTORY tables. This functionality can be extended by extending the  MOVECLS_MineMovementClassAnalysisData stored procedure or by adding a custom stored procedure via a modification.

The MOVECLS_MineMovementClassAnalysisData stored procedure is responsible for the data migration functionality, below is an example that will illustrate how the functionality can be extended.

 Modify the MOVECLS_MineMovementClassAnalysisData Stored Procedure

A new nullable or default value parameter can be added to the stored procedure to allow the SP to be called from the existing and modified code. This will prevent breaking the existing stored procedure call. Here the WorkZone nullable field has been added in the existing stored procedure.

 

MOVECLS_MineMovementClassAnalysisData
Copy Code 
ALTER PROCEDURE [dbo].[MOVECLS_MineMovementClassAnalysisData] (
      @Warehouse NVARCHAR(25),
      @Days INT = 90,
      @Permanent NCHAR(1),
      @IncludeArchivedData NCHAR(1),
      @WorkZone NVARCHAR(25) = NULL  --- 'Work Zone has been added under extensibility.'      )
AS
SET NOCOUNT ON;

DECLARE @DateFrom DATETIME
DECLARE @DateTo DATETIME
DECLARE @stErrorMsg NVARCHAR(2000)
DECLARE @rowEffected NUMERIC(9, 0)

SELECT @DateTo = DATEADD(dd, 1, DATEDIFF(dd, 0, GETDATE()))

SELECT @DateFrom = @DateTo - @Days

--Wareshouse should not be null
IF (@Warehouse IS NULL)
BEGIN
      SET @stErrorMsg = N'MSG_INVVAL28';

      RETURN - 1;
END

--Clear MOVEMENT_CLASS_ANALYSIS table on each run
DELETE
FROM MOVEMENT_CLASS_ANALYSIS
WHERE WAREHOUSE = @Warehouse

IF UPPER(@PERMANENT) = N'Y'
BEGIN
      INSERT INTO MOVEMENT_CLASS_ANALYSIS (
            ITEM,
            COMPANY,
            WAREHOUSE,
            LOCATION,
            TOTAL_QUANTITY,
            QUANTITY_UM,
            ITEM_CLASS,
            NUMBER_OF_HITS,
            NUMBER_OF_DAYS,
            DATE_TIME_STAMP,
            USER_STAMP,
            PROCESS_STAMP
            )
      SELECT ITEM,
            COMPANY,
            WAREHOUSE,
            LOCATION,
            sum(TOTAL_QUATITY) AS TOTAL_QUATITY,
            QUANTITY_UM,
            ITEM_CLASS,
            SUM(NUMBER_OF_HITS),
            @Days,
            GETDATE(),
            N'System',
            N'MineMovementClassAnalysis'
      FROM (
            SELECT TH.ITEM,
                  TH.COMPANY,
                  TH.WAREHOUSE,
                  TH.LOCATION,
                  sum(TH.QUANTITY) AS TOTAL_QUATITY,
                  th.QUANTITY_UM,
                  I.ITEM_CLASS,
                  NUMBER_OF_HITS
            FROM ITEM I WITH (NOLOCK)
            JOIN (
                  SELECT TRANSACTION_HISTORY.ITEM,
                        TRANSACTION_HISTORY.COMPANY,
                        TRANSACTION_HISTORY.WAREHOUSE,
                        TRANSACTION_HISTORY.LOCATION,
                        TRANSACTION_HISTORY.QUANTITY_UM,
                        sum(TRANSACTION_HISTORY.QUANTITY) AS QUANTITY,
                        COUNT(*) AS NUMBER_OF_HITS
                  FROM TRANSACTION_HISTORY WITH (NOLOCK)
                  INNER JOIN LOCATION WITH (NOLOCK)
                        ON LOCATION.LOCATION = TRANSACTION_HISTORY.LOCATION
                              AND LOCATION.WAREHOUSE = TRANSACTION_HISTORY.WAREHOUSE
                              AND LOCATION.LOCATION_CLASS = N'Inventory'
                  INNER JOIN LOCATION_INVENTORY LI
                        ON TRANSACTION_HISTORY.LOCATION = LI.LOCATION
                              AND TRANSACTION_HISTORY.WAREHOUSE = LI.WAREHOUSE
                  WHERE ACTIVITY_DATE_TIME >= @DateFrom
                        AND ACTIVITY_DATE_TIME < @DateTo
                        AND DIRECTION = N'FROM'
                        AND TRANSACTION_TYPE IN (
                              40,
                              60,
                              100,
                              130,
                              120,
                              240,
                              250,
                              360
                              )
                        AND TRANSACTION_HISTORY.WAREHOUSE = @Warehouse
                        AND (
                              @WorkZone IS NULL
                              OR (ISNULL(@WorkZone, '!') = ISNULL(TRANSACTION_HISTORY.WORK_ZONE, '!'))
                              ) --- 'Work Zone has been added under extensibility.'
                        AND (LI.PERMANENT <> N'N')
                  GROUP BY TRANSACTION_HISTORY.ITEM,
                        TRANSACTION_HISTORY.COMPANY,
                        TRANSACTION_HISTORY.WAREHOUSE,
                        TRANSACTION_HISTORY.LOCATION,
                        TRANSACTION_HISTORY.QUANTITY_UM
                  ) TH
                  ON I.ITEM = TH.ITEM
                        AND (
                              TH.COMPANY = I.COMPANY
                              OR (
                                    TH.COMPANY IS NULL
                                    OR I.COMPANY IS NULL
                                    )
                              )
            WHERE TH.WAREHOUSE = @WAREHOUSE
            GROUP BY TH.ITEM,
                  TH.COMPANY,
                  TH.LOCATION,
                  TH.WAREHOUSE,
                  TH.QUANTITY_UM,
                  I.ITEM_CLASS,
                  NUMBER_OF_HITS
            
            UNION
            
            SELECT TH.ITEM,
                  TH.COMPANY,
                  TH.WAREHOUSE,
                  TH.LOCATION,
                  sum(TH.QUANTITY) AS TOTAL_QUATITY,
                  th.QUANTITY_UM,
                  I.ITEM_CLASS,
                  NUMBER_OF_HITS
            FROM ITEM I WITH (NOLOCK)
            JOIN (
                  SELECT AR_TRANSACTION_HISTORY.ITEM,
                        AR_TRANSACTION_HISTORY.COMPANY,
                        AR_TRANSACTION_HISTORY.WAREHOUSE,
                        AR_TRANSACTION_HISTORY.LOCATION,
                        AR_TRANSACTION_HISTORY.QUANTITY_UM,
                        sum(AR_TRANSACTION_HISTORY.QUANTITY) AS QUANTITY,
                        COUNT(*) AS NUMBER_OF_HITS
                  FROM AR_TRANSACTION_HISTORY WITH (NOLOCK)
                  INNER JOIN LOCATION WITH (NOLOCK)
                        ON LOCATION.LOCATION = AR_TRANSACTION_HISTORY.LOCATION
                              AND LOCATION.WAREHOUSE = AR_TRANSACTION_HISTORY.WAREHOUSE
                              AND LOCATION.LOCATION_CLASS = N'Inventory'
                  INNER JOIN LOCATION_INVENTORY LI
                        ON AR_TRANSACTION_HISTORY.LOCATION = LI.LOCATION
                              AND AR_TRANSACTION_HISTORY.WAREHOUSE = LI.WAREHOUSE
                  WHERE ACTIVITY_DATE_TIME >= @DateFrom
                        AND ACTIVITY_DATE_TIME < @DateTo
                        AND DIRECTION = N'FROM'
                        AND TRANSACTION_TYPE IN (
                              40,
                              60,
                              100,
                              130,
                              120,
                              240,
                              250,
                              360
                              )
                        AND AR_TRANSACTION_HISTORY.WAREHOUSE = @Warehouse
                        AND (
                              @WorkZone IS NULL
                              OR (ISNULL(@WorkZone, '!') = ISNULL(AR_TRANSACTION_HISTORY.WORK_ZONE, '!'))
                              ) --- 'Work Zone has been added under extensibility.'
                        AND (LI.PERMANENT <> N'N')
                  GROUP BY AR_TRANSACTION_HISTORY.ITEM,
                        AR_TRANSACTION_HISTORY.COMPANY,
                        AR_TRANSACTION_HISTORY.WAREHOUSE,
                        AR_TRANSACTION_HISTORY.LOCATION,
                        AR_TRANSACTION_HISTORY.QUANTITY_UM
                  ) TH
                  ON I.ITEM = TH.ITEM
                        AND (
                              TH.COMPANY = I.COMPANY
                              OR (
                                    TH.COMPANY IS NULL
                                    OR I.COMPANY IS NULL
                                    )
                              )
            WHERE TH.WAREHOUSE = @WAREHOUSE
                  AND @IncludeArchivedData = N'Y'
            GROUP BY TH.ITEM,
                  TH.COMPANY,
                  TH.LOCATION,
                  TH.WAREHOUSE,
                  TH.QUANTITY_UM,
                  I.ITEM_CLASS,
                  NUMBER_OF_HITS
            ) TransactionHistoryData
      GROUP BY ITEM,
            COMPANY,
            LOCATION,
            WAREHOUSE,
            QUANTITY_UM,
            ITEM_CLASS
END
ELSE
BEGIN
      INSERT INTO MOVEMENT_CLASS_ANALYSIS (
            ITEM,
            COMPANY,
            WAREHOUSE,
            LOCATION,
            TOTAL_QUANTITY,
            QUANTITY_UM,
            ITEM_CLASS,
            NUMBER_OF_HITS,
            NUMBER_OF_DAYS,
            DATE_TIME_STAMP,

            )
      SELECT ITEM,

      FROM (

                  sum(TH.QUANTITY) AS TOTAL_QUATITY,

            FROM ITEM I WITH (NOLOCK)
            JOIN (

                        TRANSACTION_HISTORY.LOCATION,

                  FROM TRANSACTION_HISTORY WITH (NOLOCK)
                  INNER JOIN LOCATION WITH (NOLOCK)
                        ON LOCATION.LOCATION = TRANSACTION_HISTORY.LOCATION
                              AND LOCATION.WAREHOUSE = TRANSACTION_HISTORY.WAREHOUSE
                              AND LOCATION.LOCATION_CLASS = N'Inventory'
                  WHERE ACTIVITY_DATE_TIME >= @DateFrom
                        AND ACTIVITY_DATE_TIME < @DateTo
                        AND DIRECTION = N'FROM'
                        AND TRANSACTION_TYPE IN (
                              40,

                              100,

                              )
                        AND TRANSACTION_HISTORY.WAREHOUSE = @Warehouse
                        AND (
                              @WorkZone IS NULL
                              OR (ISNULL(@WorkZone, '!') = ISNULL(TRANSACTION_HISTORY.WORK_ZONE, '!'))
                              )  --- 'Work Zone has been added under extensibility.'
                  GROUP BY TRANSACTION_HISTORY.ITEM,
                        TRANSACTION_HISTORY.COMPANY,
                        TRANSACTION_HISTORY.WAREHOUSE,
                        TRANSACTION_HISTORY.LOCATION,
                        QUANTITY_UM
                  ) TH
                  ON I.ITEM = TH.ITEM
                        AND (
                              TH.COMPANY = I.COMPANY
                              OR (
                                    TH.COMPANY IS NULL
                                    OR I.COMPANY IS NULL
                                    )
                              )
            WHERE TH.WAREHOUSE = @WAREHOUSE
            GROUP BY TH.ITEM,
                  TH.COMPANY,
                  TH.LOCATION,
                  TH.WAREHOUSE,
                  TH.QUANTITY_UM,
                  I.ITEM_CLASS,

            UNION
            
            SELECT TH.ITEM,
                  TH.COMPANY,
                  TH.WAREHOUSE,

                  sum(TH.QUANTITY) AS TOTAL_QUATITY,

            FROM ITEM I WITH (NOLOCK)
            JOIN (

                        AR_TRANSACTION_HISTORY.WAREHOUSE,

                        AND ACTIVITY_DATE_TIME < @DateTo

                              40,

                              130,

 Extend MOVEMENT_CLASS_ANALYSIS_VIEW View

The new WORK_ZONE column added above can be added to the view MOVEMENT_CLASS_ANALYSIS_VIEW, which is the Header Data Source for the viewer. The example below will show this modification  

MOVEMENT_CLASS_ANALYSIS_VIEW
Copy Code 
ALTER VIEW MOVEMENT_CLASS_ANALYSIS_VIEW 
AS
SELECT 
    ISNULL(
        (
            CASE 
                WHEN (
                ( 
                    SELECT  COUNT(GC.OBJECT_ID) 
                    FROM GENERIC_CONFIG_DETAIL AS GC WITH (NOLOCK) 
                    INNER JOIN ITEM_UNIT_OF_MEASURE AS IU WITH(NOLOCK) 
                        ON GC.IDENTIFIER=IU.ITEM_CLASS
                ) <> 0) 
                THEN ITEM_INFO.ITEMCLASSOBJECTID 
            END
        ),
        
    ITEM_INFO.INTERNALITEMNUM ) AS OBJECT_ID,
    MCA.ITEM,ITEM_INFO.ITEMDESC AS DESCRIPTION,
    ITEM_INFO.ITEMCLASS AS ITEM_CLASS,
    MCA.COMPANY,
    MCA.LOCATION,
    L.LOCATION_TYPE,
    L.LOCATING_ZONE,
    MCA.WAREHOUSE,
    NUMBER_OF_HITS,
    MCA.TOTAL_QUANTITY,
    MCA.QUANTITY_UM,
    MCA.NUMBER_OF_DAYS,
    MCA.DATE_TIME_STAMP,
    ITEM_INFO.VALUEPERITEM * MCA.TOTAL_QUANTITY AS TOTAL_VALUE_PICKED,
    ITEM_INFO.COSTPERITEM * MCA.TOTAL_QUANTITY AS TOTAL_COST_PICKED,
    CONVERSION_INFO.WEIGHTPERITEM * (MCA.TOTAL_QUANTITY/CONVERSION_INFO.CONVERSIONQTY) AS TOTAL_WEIGHT_PICKED,
    ITEM_INFO.MOVEMENTCLASS AS QUANTITY_UM_MOVEMENT_CLASS,
    L.MOVEMENT_CLS AS LOCATION_MOVEMENT_CLASS, 
    CASE 
        WHEN LEN(ILA.ALLOCATION_LOC) > 0 
        THEN 'Y' 
        ELSE 'N' 
    END AS PERMANENT,
    CASE 
        WHEN CONVERSION_INFO.CONVERSIONQUANTITYUM != MCA.QUANTITY_UM 
        THEN MCA.TOTAL_QUANTITY/CONVERSION_INFO.CONVERSIONQTY 
        ELSE MCA.TOTAL_QUANTITY 
    END AS CONVERTED_QTY,
    CONVERSION_INFO.CONVERSIONQUANTITYUM AS CONVERTED_QTY_UM, 
    CONVERSION_INFO.CONVERSIONQUANTITYMOVEMENTCLASS AS CONVERTED_UM_MOVEMENT_CLASS,
    L.WORK_ZONE  --- 'Work Zone has been added under extensibility.'
FROM 
    MOVEMENT_CLASS_ANALYSIS MCA WITH (NOLOCK) 
    LEFT OUTER JOIN ITEM_LOCATION_ASSIGNMENT ILA WITH (NOLOCK) 
        ON ILA.WAREHOUSE = MCA.WAREHOUSE
            AND ILA.ALLOCATION_LOC= MCA.LOCATION 
            AND ILA.ITEM = MCA.ITEM 
            AND (ILA.COMPANY = MCA.COMPANY OR (ILA.COMPANY IS NULL AND MCA.COMPANY IS NULL)) 
    LEFT OUTER JOIN LOCATION L WITH (NOLOCK) 
        ON L.LOCATION = MCA.LOCATION 
            AND L.WAREHOUSE = MCA.WAREHOUSE
    OUTER APPLY INVFN_RTRVITEMINFOFORBASEUM(
            MCA.ITEM,
            MCA.COMPANY,
            MCA.WAREHOUSE
            ) AS ITEM_INFO 
    OUTER APPLY INVfn_RtrvConversionInfoForItemAndLocation(
        MCA.ITEM,
        MCA.COMPANY,
        MCA.WAREHOUSE,
        MCA.LOCATION,
        MCA.TOTAL_QUANTITY,
        MCA.QUANTITY_UM,
        ITEM_INFO.MOVEMENTCLASS, 
        ITEM_INFO.weightPerItem
        ) AS CONVERSION_INFO;
