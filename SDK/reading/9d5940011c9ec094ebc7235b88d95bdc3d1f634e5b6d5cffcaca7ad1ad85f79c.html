Integration Endpoints > Rating Service Endpoints > Get Manifests

The Get Manifests Rating Service Action is invoked from the Manifest Insight.

Make sure that the request and response messages adhere to the following schema:

  • GetManifestsRequest.xsd
    • Parcel.xsd
    • UserDefinedFields.xsd
  • GetManifestsResponse.xsd
    • Result.xsd
    • Manifest.xsd



When utilizing an API style Integration Endpoint for this action, make sure the configured class and method has the following form:

 

C#
Copy Code 
/// <summary>
/// Class that defines the GetManifests Endpoint
/// </summary>
public class GetManifests : IWorkFlowStep
{
    /// <summary>
    /// Method that is executed for the GetManifests Endpoint
    /// </summary>
    /// <param name="session">
    /// Manh.WMFW.General.Session of the WMW.General assembly.
    /// </param>
    /// <param name="parameters">
    /// object that holds the serialized request of the endpoint
    /// </param>
    /// <returns>
    /// Serialized response from the endpoint.
    /// </returns>
    public object ExecuteStep(Session session, params object[] parameters)
    {
        string requestMsg = Convert.ToString(parameters[0]);
        GetManifestsRequest request = (GetManifestsRequest)XMLDocManager.Deserialize(typeof(GetManifestsRequest), requestMsg);
        GetManifestsResponse response = new GetManifestsResponse();
        response.Result = new Manh.ILS.General.General.Result();

        if(request == null)
        {
            response.Result.SuccessFlag = false;
            response.Result.ErrorMessage = "No request message found";
        }
        else
        {
            //validate against response schema

            //If schema fails
            {
                response.Result.SuccessFlag = false;
                response.Result.ErrorMessage = "Error message goes here";
            }
            else
            {
                //process request
                //set true success or false on response result based on processing
            }   
        }

        //Return the serialized response
        return XMLDocManager.Serialize(response, null);
    }
}

public static string FetchTagValue(XmlDocument xmlDoc, string TagName)
{
  for(var i = 0; i < xmlDoc.GetElementsByTagName(TagName).Count; i++)
  {
        var innerValue = xmlDoc.GetElementsByTagName(TagName)[i].InnerXml;
        if(!string.IsNullOrEmpty(innerValue)) return innerValue;
  }
  return string.Empty;
}
