Integration Endpoints > Rating Service Endpoints > Get Containers

The Get Containers Rating Service Action is invoked from the Manifest Insight.

Make sure that the request and response messages adhere to the following schema:

  • GetContainersRequest.xsd
    • Manifest.xsd
    • UserDefinedFields.xsd
  • GetContainersResponse.xsd
    • Result.xsd
    • Parcel.xsd


When utilizing an API style Integration Endpoint for this action, make sure the configured class and method has the following form:

 

Example Title
Copy Code 
/// <summary>
/// Class that defines the CloseManifest Endpoint
/// </summary>
public class GetContainers : IWorkFlowStep
{
    /// <summary>
    /// Method that is executed for the GetContainers Endpoint
    /// </summary>
    /// <param name="session">
    /// Manh.WMFW.General.Session of the WMW.General assembly.
    /// </param>
    /// <param name="parameters">
    /// object that holds the serialized request of the endpoint
    /// </param>
    /// <returns>
    /// Serialized response from the endpoint.
    /// </returns>
    public object ExecuteStep(Session session, params object[] parameters)
    {
        string requestMsg = Convert.ToString(parameters[0]);
        GetContainersRequest request = (GetContainersRequest)XMLDocManager.Deserialize(typeof(GetContainersRequest), requestMsg);
        GetContainersResponse response = new GetContainersResponse();
        response.Result = new Manh.ILS.General.General.Result();

        if(request == null)
        {
            response.Result.SuccessFlag = false;
            response.Result.ErrorMessage = "No request message found";
        }
        else
        {
            //validate against response schema

            //If schema fails
            {
                response.Result.SuccessFlag = false;
                response.Result.ErrorMessage = "Error message goes here";
            }
            else
            {
                //process request
                //set true success or false on response result based on processing
            }   
        }

        //Return the serialized response
        return XMLDocManager.Serialize(response, null);
    }
}
