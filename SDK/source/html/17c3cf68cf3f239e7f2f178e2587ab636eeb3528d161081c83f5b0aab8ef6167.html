Containers can be manifested from the Shipping container insight screen.

Make sure that the request and response messages adhere to the following schemas:

  • ManifestContainerRequest.xsd
    • Accessorial.xsd
    • Address.xsd
    • UserDefinedFields.xsd
  • ManifestContainerResponse.xsd
    • Parcel.xsd
    • Result.xsd
    • UserDefinedFields.xsd



When utilizing an API style Integration Endpoint for this action, make sure the configured class and method has the following form:

C#
Copy Code 
/// <summary>
/// Class that defines the ManifestParcel Endpoint
/// </summary>
public class ManifestParcel : IWorkFlowStep
{
    /// <summary>
    /// Method that is executed for the ManifestParcel Endpoint
    /// </summary>
    /// <param name="session">
    /// Manh.WMFW.General.Session of the WMW.General assembly.
    /// </param>
    /// <param name="parameters">
    /// object that holds the serialized request of the endpoint
    /// </param>
    /// <returns>
    /// Serialized response from the endpoint.
    /// </returns>
    public object ExecuteStep(Session session, params object[] parameters)
    {
        string requestMsg = Convert.ToString(parameters[0]);
        ManifestParcelRequest request = (ManifestParcelRequest)XMLDocManager.Deserialize(typeof(ManifestParcelRequest), requestMsg);
        ManifestParcelResponse response = new ManifestParcelResponse();
        response.Result = new Manh.ILS.General.General.Result();

        if(request == null)
        {
            response.Result.SuccessFlag = false;
            response.Result.ErrorMessage = "No request message found";
        }
        else
        {
            //validate against response schema

            //If schema fails
            {
                response.Result.SuccessFlag = false;
                response.Result.ErrorMessage = "Error message goes here";
            }
            else
            {
                //process request
                //set true success or false on response result based on processing
            }   
        }

        //Return the serialized response
        return XMLDocManager.Serialize(response, null);
    }
}

When utilizing web service style Integration Endpoint for this action, make sure the configured web service accepts request and returns response in the above mentioned XSD formats.

using System.Net;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Primitives;
using Newtonsoft.Json;
using System.Xml;
using System;
public static string BuildParcel(XmlDocument xmlDoc){
    var parcel = "";
    for(var i = 0; i < xmlDoc.GetElementsByTagName("ContainerId").Count; i++){
        var containerId = xmlDoc.GetElementsByTagName("ContainerId")[i].InnerXml;
        var internalContNum = xmlDoc.GetElementsByTagName("InternalContainerNum")[i].InnerXml;
        parcel = parcel +
        "<Parcel>"+
        "      <ContainerId xmlns=\"http://www.manh.com/ILSNET/Shipping/Parcel\">"+containerId+"</ContainerId>"+
        "      <InternalContainerNumber xmlns=\"http://www.manh.com/ILSNET/Shipping/Parcel\">"+internalContNum+"</InternalContainerNumber>"+
        "      <MSN xmlns=\"http://www.manh.com/ILSNET/Shipping/Parcel\">"+internalContNum+"</MSN>"+                                                                
        "      <TrackingNumber xmlns=\"http://www.manh.com/ILSNET/Shipping/Parcel\">"+"TN-" + containerId+"-"+System.DateTime.Now.ToString("HH:mm:ss")+"</TrackingNumber>"+
        " <ShippingContainerUserDefinedFields xmlns=\"http://www.manh.com/ILSNET/Shipping/Parcel\">"+
        " </ShippingContainerUserDefinedFields>"+
        "</Parcel>";
    }
    return parcel;
}
