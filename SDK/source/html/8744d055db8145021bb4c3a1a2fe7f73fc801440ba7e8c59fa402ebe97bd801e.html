Containers can be unmanifested from the Shipping container insight.

Make sure that the request and response messages adhere to the following schemas:

  • UnManifestContainerRequest.xsd
    • Parcel.xsd
    • UserDefinedFields.xsd
  • UnManifestContainerResponse.xsd
    • Result.xsd


When utilizing an API style Integration Endpoint for this action, make sure the configured class and method has the following form:

C#
Copy Code 
/// <summary>
/// Class that defines the UnManifestContainer Endpoint
/// </summary>
public class UnManifestContainer : IWorkFlowStep
{
    /// <summary>
    /// Method that is executed for the UnManifestContainer Endpoint
    /// </summary>
    /// <param name="session">
    /// Manh.WMFW.General.Session of the WMW.General assembly.
    /// </param>
    /// <param name="parameters">
    /// object that holds the serialized request of the endpoint
    /// </param>
    /// <returns>
    /// Serialized response from the endpoint.
    /// </returns>
    public object ExecuteStep(Session session, params object[] parameters)
    {
        string requestMsg = Convert.ToString(parameters[0]);
        UnManifestContainerRequest request = (UnManifestContainerRequest)XMLDocManager.Deserialize(typeof(UnManifestContainerRequest), requestMsg);
        UnManifestContainerResponse response = new UnManifestContainerResponse();
        response.Result = new Manh.ILS.General.General.Result();

        if(request == null)
        {
            response.Result.SuccessFlag = false;
            response.Result.ErrorMessage = "No request message found";
        }
        else
        {
            //validate against response schema

            //If schema fails
            {
                response.Result.SuccessFlag = false;
                response.Result.ErrorMessage = "Error message goes here";
            }
            else
            {
                //process request
                //set true success or false on response result based on processing
            }   
        }

        //Return the serialized response
        return XMLDocManager.Serialize(response, null);
    }
}

When utilizing web service style Integration Endpoint for this action, make sure the configured web service accepts request and returns response in the above mentioned XSD formats.

Below is an example of an Azure function for the same :

#r "Newtonsoft.Json"
using System.Net;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Primitives;
using Newtonsoft.Json;
public static async Task<IActionResult> Run(HttpRequest req, ILogger log)
{
log.LogInformation("C# HTTP trigger function processed a request.");
string name = req.Query["name"];
string requestBody = await new StreamReader(req.Body).ReadToEndAsync();
dynamic data = JsonConvert.DeserializeObject(requestBody);
name = name ?? data?.name;
string response = "<?xml version = \"1.0\" encoding = \"utf-8\" ?>" +
"<UnmanifestContainerResponse xmlns =\"http://www.manh.com/ILSNET/Shipping/UnmanifestContainerResponse\" >" +
"<Result>" +
"<SuccessFlag xmlns =\"http://www.manh.com/ILSNET/General/Result\" >true</SuccessFlag>" +
"<ErrorMessage xmlns =\"http://www.manh.com/ILSNET/General/Result\"></ErrorMessage>" +
"</Result>" +
"</UnmanifestContainerResponse>";
return new OkObjectResult(response);
}
